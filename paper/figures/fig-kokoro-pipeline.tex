% fig-kokoro-pipeline.tex
% CS1 (Kokoro-82M) decomposed inference pipeline, shipped placement.
% Data: paper/main.tex Section 4.1 and Table tab:cs1-placement (shipped
%       placement per stage), Section sec:cs1-hybrid (hybrid generator:
%       53 ANE convolution functions, AdaIN/Snake/joins in Metal, IOSurface
%       shared memory, >=7 s; folded all-ANE below 7 s and as fallback).
% Four Core ML model families, three native Swift stages. Per-stage placement:
%   Duration  = BERT front Core ML .cpuAndNeuralEngine + LSTM back Core ML .cpuOnly
%   Alignment = Swift (CPU)
%   F0Ntrain  = front Core ML .cpuOnly + back Core ML .cpuAndNeuralEngine
%   Pad       = host materialization
%   DecoderPre = Core ML .cpuAndNeuralEngine, concurrent with hn-nsf (Swift/vDSP)
%   Generator = hybrid ANE convolutions + Metal (GPU) elementwise (>=7 s);
%               folded all-ANE below 7 s and as fallback
%   iSTFT tail + Trim = Swift (CPU)
% Build: tectonic paper/figures/fig-kokoro-pipeline.tex
\documentclass[border=6pt]{standalone}

% Okabe-Ito-derived, colorblind-safe fills; distinct border dash = redundant
% (grayscale-safe) encoding of engine placement.
\definecolor{cmlcpu}{HTML}{E0E0D1}   % Core ML on CPU       (neutral khaki)
\definecolor{cmlane}{HTML}{C7E9C0}   % Core ML on CPU+ANE   (green family, the ANE)
\definecolor{metalg}{HTML}{CDE4F2}   % Metal on GPU         (blue family)
\definecolor{swiftc}{HTML}{FDE3C4}   % native Swift (CPU)   (orange family)
\definecolor{padc}{HTML}{DCD6E8}     % host materialization (padding) -- not a named Swift stage; distinct from ioblk
\definecolor{ioblk}{HTML}{E8E8E8}    % I/O endpoints
\definecolor{edgec}{HTML}{444444}

  {\textbf{iSTFT tail $+$ Trim}};
